```
impl Algorithm {
    /// Convert a string from the input file into an Algorithm enum.
    ///
    /// Centralizing this conversion avoids scattering string comparisons
    /// throughout the parser and ensures that any future algorithms are
    /// added in a single, maintainable location.
    fn from_str(value: &str) -> Result<Self, String> {
        match value {
            "fcfs" => Ok(Algorithm::FCFS),
            "sjf" => Ok(Algorithm::SJF),
            "rr" => Ok(Algorithm::RR),
            _ => Err(format!("Invalid scheduling algorithm '{}'", value)),
        }
    }

    fn name(&self) -> &'static str {
        match self {
            Algorithm::FCFS => "FCFS",
            Algorithm::SJF => "SJF",
            Algorithm::RR => "RR",
        }
    }
}
```
